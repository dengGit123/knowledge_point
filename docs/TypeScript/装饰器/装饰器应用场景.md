# 装饰器应用场景

## 一、概述

装饰器在实际开发中有广泛的应用场景，尤其在框架开发中。本文介绍几种最常见的应用模式。

## 二、AOP（面向切面编程）

AOP 是装饰器最经典的应用，用于将**横切关注点**（如日志、权限、事务）从业务逻辑中分离。

### 1. 日志记录

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    console.log(`[${new Date().toISOString()}] → ${String(key)}(${args.join(', ')})`);
    const result = original.apply(this, args);
    console.log(`[${new Date().toISOString()}] ← ${String(key)} 返回: ${result}`);
    return result;
  };
}

class OrderService {
  @Log
  createOrder(productId: string, quantity: number) {
    return `订单创建成功: ${productId} x${quantity}`;
  }

  @Log
  cancelOrder(orderId: string) {
    return `订单 ${orderId} 已取消`;
  }
}
```

### 2. 性能监控

```typescript
function Measure(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const start = performance.now();
    const result = original.apply(this, args);
    const duration = performance.now() - start;

    // 上报性能数据
    console.log(`⏱ ${String(key)} 耗时: ${duration.toFixed(2)}ms`);

    // 可以上报到监控系统
    // reportMetric({ method: String(key), duration });

    return result;
  };
}

class DataService {
  @Measure
  fetchLargeDataset() {
    // 模拟耗时操作
    return Array.from({ length: 10000 }, (_, i) => i);
  }
}
```

### 3. 错误处理

```typescript
function CatchError(fallback?: any) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = function (...args: any[]) {
      try {
        return original.apply(this, args);
      } catch (error) {
        console.error(`❌ ${String(key)} 执行出错:`, error);
        return fallback;
      }
    };
  };
}

class ApiService {
  @CatchError({ error: '请求失败' })
  fetchData(url: string) {
    // 可能抛出错误
    if (!url) throw new Error('URL 不能为空');
    return `数据来自 ${url}`;
  }
}
```

## 三、依赖注入（DI）

装饰器是实现依赖注入的核心机制，常见于 Angular、NestJS 等框架。

### 1. 简单 DI 容器

```typescript
// 服务容器
const Container = new Map<string, any>();

// 注册服务
function Injectable(token: string) {
  return function (constructor: Function) {
    Container.set(token, constructor);
  };
}

// 注入服务
function Inject(token: string) {
  return function (target: any, propertyKey: string) {
    Object.defineProperty(target, propertyKey, {
      get() {
        const Service = Container.get(token);
        if (!Service) {
          throw new Error(`服务 ${token} 未注册`);
        }
        // 单例模式
        if (!Service._instance) {
          Service._instance = new Service();
        }
        return Service._instance;
      },
      enumerable: true,
      configurable: true
    });
  };
}

// 使用
@Injectable('logger')
class Logger {
  log(message: string) {
    console.log(`[LOG] ${message}`);
  }
}

@Injectable('userService')
class UserService {
  @Inject('logger')
  logger!: Logger;

  getUsers() {
    this.logger.log('获取用户列表');
    return ['Alice', 'Bob'];
  }
}

class UserController {
  @Inject('userService')
  userService!: UserService;

  listUsers() {
    return this.userService.getUsers();
  }
}

// 使用
const controller = new UserController();
console.log(controller.listUsers());
// 输出: [LOG] 获取用户列表
//      ['Alice', 'Bob']
```

```typescript
// 分层架构示例
const services = new Map();

function Service(name: string) {
  return function (constructor: Function) {
    services.set(name, constructor);
  };
}

function Repository(name: string) {
  return function (constructor: Function) {
    services.set(name, constructor);
  };
}

function Autowired(name: string) {
  return function (target: any, propertyKey: string) {
    Object.defineProperty(target, propertyKey, {
      get() {
        const Cls = services.get(name);
        return Cls ? new Cls() : null;
      }
    });
  };
}

@Repository('userRepository')
class UserRepository {
  findById(id: number) {
    return { id, name: `User${id}` };
  }
}

@Service('userService')
class UserService {
  @Autowired('userRepository')
  repository!: UserRepository;

  getUser(id: number) {
    return this.repository.findById(id);
  }
}

class UserController {
  @Autowired('userService')
  service!: UserService;

  getUser(id: number) {
    return this.service.getUser(id);
  }
}
```

## 四、路由注册

Web 框架中使用装饰器注册路由是常见模式。

### 1. 路由装饰器

```typescript
// 路由注册表
interface Route {
  method: string;
  path: string;
  handler: Function;
  middlewares?: Function[];
}

const routes: Route[] = [];

// HTTP 方法装饰器
function Get(path: string, ...middlewares: Function[]) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    routes.push({ method: 'GET', path, handler: descriptor.value, middlewares });
  };
}

function Post(path: string, ...middlewares: Function[]) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    routes.push({ method: 'POST', path, handler: descriptor.value, middlewares });
  };
}

function Put(path: string) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    routes.push({ method: 'PUT', path, handler: descriptor.value });
  };
}

function Delete(path: string) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    routes.push({ method: 'DELETE', path, handler: descriptor.value });
  };
}

// 控制器装饰器
function Controller(basePath: string) {
  return function (constructor: Function) {
    constructor.prototype._basePath = basePath;
  };
}

// 中间件
function Auth(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log('🔐 验证身份...');
    return original.apply(this, args);
  };
}

// 使用
@Controller('/users')
class UserController {
  @Get('/')
  findAll() {
    return '获取所有用户';
  }

  @Get('/:id')
  findOne(id: string) {
    return `获取用户 ${id}`;
  }

  @Post('/')
  @Auth
  create() {
    return '创建用户';
  }

  @Put('/:id')
  update(id: string) {
    return `更新用户 ${id}`;
  }

  @Delete('/:id')
  remove(id: string) {
    return `删除用户 ${id}`;
  }
}

console.log(routes);
```

### 1. 表单验证

```typescript
// 验证装饰器
function MinLength(min: number) {
  return function (target: any, propertyKey: string) {
    let value: string = '';
    Object.defineProperty(target, propertyKey, {
      get() { return value; },
      set(newValue: string) {
        if (newValue.length < min) {
          throw new Error(`${propertyKey} 至少需要 ${min} 个字符`);
        }
        value = newValue;
      }
    });
  };
}

function MaxLength(max: number) {
  return function (target: any, propertyKey: string) {
    let value: string = '';
    Object.defineProperty(target, propertyKey, {
      get() { return value; },
      set(newValue: string) {
        if (newValue.length > max) {
          throw new Error(`${propertyKey} 最多 ${max} 个字符`);
        }
        value = newValue;
      }
    });
  };
}

function Pattern(regex: RegExp, message: string) {
  return function (target: any, propertyKey: string) {
    let value: string = '';
    Object.defineProperty(target, propertyKey, {
      get() { return value; },
      set(newValue: string) {
        if (!regex.test(newValue)) {
          throw new Error(message);
        }
        value = newValue;
      }
    });
  };
}

class UserForm {
  @MinLength(3)
  @MaxLength(20)
  username!: string;

  @Pattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/, '邮箱格式不正确')
  email!: string;

  @MinLength(6)
  @MaxLength(30)
  password!: string;
}

const form = new UserForm();
form.username = 'Alice'; // OK
form.email = 'alice@example.com'; // OK
// form.email = 'invalid'; // 报错
```

### 2. DTO 验证

```typescript
// 验证元数据
const validationRules = new WeakMap<object, Map<string, Function[]>>();

function IsString(target: any, propertyKey: string) {
  addValidation(target, propertyKey, (value: any) => {
    if (typeof value !== 'string') throw new Error(`${propertyKey} 必须是字符串`);
  });
}

function IsNotEmpty(target: any, propertyKey: string) {
  addValidation(target, propertyKey, (value: any) => {
    if (!value || value.trim() === '') throw new Error(`${propertyKey} 不能为空`);
  });
}

function IsEmail(target: any, propertyKey: string) {
  addValidation(target, propertyKey, (value: any) => {
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
      throw new Error(`${propertyKey} 必须是有效的邮箱`);
    }
  });
}

function Min(min: number) {
  return function (target: any, propertyKey: string) {
    addValidation(target, propertyKey, (value: any) => {
      if (typeof value === 'string' && value.length < min) {
        throw new Error(`${propertyKey} 长度不能少于 ${min}`);
      }
      if (typeof value === 'number' && value < min) {
        throw new Error(`${propertyKey} 不能小于 ${min}`);
      }
    });
  };
}

function addValidation(target: any, propertyKey: string, validator: Function) {
  let rules = validationRules.get(target);
  if (!rules) {
    rules = new Map();
    validationRules.set(target, rules);
  }
  let propertyRules = rules.get(propertyKey);
  if (!propertyRules) {
    propertyRules = [];
    rules.set(propertyKey, propertyRules);
  }
  propertyRules.push(validator);
}

// 验证器
function validate(obj: any): { valid: boolean; errors: string[] } {
  const errors: string[] = [];
  const rules = validationRules.get(Object.getPrototypeOf(obj));

  if (!rules) return { valid: true, errors: [] };

  for (const [property, validators] of rules) {
    for (const validator of validators) {
      try {
        validator(obj[property]);
      } catch (error) {
        errors.push((error as Error).message);
      }
    }
  }

  return { valid: errors.length === 0, errors };
}

// 使用
class CreateUserDto {
  @IsString
  @IsNotEmpty
  name!: string;

  @IsString
  @IsNotEmpty
  @IsEmail
  email!: string;

  @Min(0)
  age!: number;
}

const dto = new CreateUserDto();
dto.name = 'Alice';
dto.email = 'alice@example.com';
dto.age = 25;

const result = validate(dto);
console.log(result); // { valid: true, errors: [] }
```

## 六、权限控制

```typescript
// 当前用户（模拟）
let currentUser: { id: string; role: string } | null = null;

function setCurrentUser(user: { id: string; role: string }) {
  currentUser = user;
}

// 权限装饰器
function RequireAuth(roles?: string[]) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = function (...args: any[]) {
      // 检查是否登录
      if (!currentUser) {
        throw new Error('请先登录');
      }

      // 检查角色权限
      if (roles && roles.length > 0 && !roles.includes(currentUser.role)) {
        throw new Error(`需要以下角色之一: ${roles.join(', ')}`);
      }

      return original.apply(this, args);
    };
  };
}

// 资源所有权检查
function RequireOwner(getResourceOwnerId: (...args: any[]) => string) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = function (...args: any[]) {
      if (!currentUser) throw new Error('请先登录');

      const ownerId = getResourceOwnerId(...args);
      if (currentUser.role !== 'admin' && currentUser.id !== ownerId) {
        throw new Error('无权访问此资源');
      }

      return original.apply(this, args);
    };
  };
}

// 使用
class UserService {
  @RequireAuth()
  getProfile() {
    return `用户资料: ${currentUser?.id}`;
  }

  @RequireAuth(['admin'])
  deleteUser(userId: string) {
    return `删除用户 ${userId}`;
  }

  @RequireAuth(['admin', 'moderator'])
  banUser(userId: string) {
    return `封禁用户 ${userId}`;
  }

  @RequireOwner((userId: string) => userId)
  updateProfile(userId: string, data: any) {
    return `更新用户 ${userId} 资料`;
  }
}

// 测试
const service = new UserService();

setCurrentUser({ id: 'user1', role: 'user' });
console.log(service.getProfile()); // OK
// service.deleteUser('user2'); // 报错：需要 admin 角色

setCurrentUser({ id: 'admin1', role: 'admin' });
console.log(service.deleteUser('user1')); // OK
```

## 七、缓存策略

```typescript
// 高级缓存装饰器
function Cached(options?: { ttl?: number; keyGenerator?: (...args: any[]) => string }) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;
    const cache = new Map<string, { value: any; expires: number }>();

    descriptor.value = function (...args: any[]) {
      const cacheKey = options?.keyGenerator
        ? options.keyGenerator(...args)
        : `${String(key)}_${JSON.stringify(args)}`;

      // 检查缓存
      const cached = cache.get(cacheKey);
      if (cached && Date.now() < cached.expires) {
        console.log(`[缓存命中] ${cacheKey}`);
        return cached.value;
      }

      // 执行原始方法
      const result = original.apply(this, args);

      // 存储缓存
      cache.set(cacheKey, {
        value: result,
        expires: options?.ttl ? Date.now() + options.ttl : Infinity
      });

      return result;
    };

    // 添加缓存管理方法
    (descriptor.value as any).clearCache = () => cache.clear();
    (descriptor.value as any).deleteCache = (...args: any[]) => {
      const cacheKey = options?.keyGenerator
        ? options.keyGenerator(...args)
        : `${String(key)}_${JSON.stringify(args)}`;
      cache.delete(cacheKey);
    };
  };
}

class ProductService {
  @Cached({ ttl: 60000 }) // 缓存 1 分钟
  getProduct(id: string) {
    console.log(`从数据库获取产品 ${id}`);
    return { id, name: `Product ${id}`, price: 99.9 };
  }

  @Cached({
    ttl: 300000,
    keyGenerator: (category: string, page: number) => `products_${category}_${page}`
  })
  getProductsByCategory(category: string, page: number) {
    console.log(`获取分类 ${category} 第 ${page} 页`);
    return [{ id: '1', name: 'Product 1' }];
  }
}
```

```typescript
// 模拟 ORM 装饰器
const entityMetadata = new WeakMap<object, any>();

function Entity(tableName: string) {
  return function (constructor: Function) {
    entityMetadata.set(constructor, {
      tableName,
      columns: []
    });
  };
}

function Column(options?: { type?: string; primary?: boolean; nullable?: boolean }) {
  return function (target: any, propertyKey: string) {
    const constructor = target.constructor;
    const meta = entityMetadata.get(constructor) || { columns: [] };
    meta.columns.push({
      name: propertyKey,
      type: options?.type || 'string',
      primary: options?.primary || false,
      nullable: options?.nullable || true
    });
    entityMetadata.set(constructor, meta);
  };
}

function PrimaryGeneratedColumn() {
  return Column({ type: 'int', primary: true });
}

// 使用
@Entity('users')
class User {
  @PrimaryGeneratedColumn()
  id!: number;

  @Column({ type: 'varchar', nullable: false })
  name!: string;

  @Column({ type: 'varchar', nullable: false })
  email!: string;

  @Column({ type: 'int' })
  age!: number;
}
```

## 九、事件驱动

```typescript
// 事件监听器
const eventHandlers = new Map<string, Function[]>();

function OnEvent(eventName: string) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    if (!eventHandlers.has(eventName)) {
      eventHandlers.set(eventName, []);
    }
    eventHandlers.get(eventName)!.push(descriptor.value);
  };
}

// 事件触发器
function emit(eventName: string, ...args: any[]) {
  const handlers = eventHandlers.get(eventName) || [];
  return handlers.map(handler => handler(...args));
}

// 使用
class OrderEvents {
  @OnEvent('order.created')
  sendConfirmationEmail(order: any) {
    console.log(`发送确认邮件给 ${order.userId}`);
  }

  @OnEvent('order.created')
  updateInventory(order: any) {
    console.log(`更新库存: ${order.productId}`);
  }

  @OnEvent('order.cancelled')
  refundPayment(order: any) {
    console.log(`退款: ${order.amount}`);
  }
}

// 触发事件
emit('order.created', { userId: 'U001', productId: 'P001', amount: 99.9 });
// 输出:
// 发送确认邮件给 U001
// 更新库存: P001
```
