# 方法装饰器（Method Decorator）

## 一、定义

**方法装饰器**是应用于类方法的装饰器，在方法声明之前声明，用来**观察、修改或替换**方法的定义。

方法装饰器在类原型上操作，可以拦截和修改方法的调用行为。

## 二、基本语法

```typescript
/**
 * 方法装饰器参数：
 * @param target - 对于静态方法是类的构造函数，对于实例方法是类的原型
 * @param propertyKey - 方法名称
 * @param descriptor - 属性描述符
 */
function Log(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalMethod = descriptor.value;

  descriptor.value = function (...args: any[]) {
    console.log(`调用方法 ${propertyKey}，参数：`, args);
    const result = originalMethod.apply(this, args);
    console.log(`方法 ${propertyKey} 返回：`, result);
    return result;
  };
}

class Calculator {
  @Log
  add(a: number, b: number): number {
    return a + b;
  }
}

const calc = new Calculator();
calc.add(1, 2);
// 输出：
// 调用方法 add，参数：[1, 2]
// 方法 add 返回：3
```

### 参数说明

| 参数 | 类型 | 说明 |
|-----|------|------|
| `target` | `any` | 实例方法 → 类的原型对象；静态方法 → 类的构造函数 |
| `propertyKey` | `string \| symbol` | 被装饰的方法名称 |
| `descriptor` | `PropertyDescriptor` | 方法的属性描述符（value、writable、enumerable、configurable） |

## 三、方法装饰器的返回值

方法装饰器可以有以下三种返回值：

### 1. 无返回值（void）

直接修改 descriptor 对象：

```typescript
function Readonly(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  descriptor.writable = false;
}

class Example {
  @Readonly
  method() {
    return '原始实现';
  }
}

const obj = new Example();
// obj.method = () => '覆盖'; // 报错：Cannot assign to read only property
```

### 2. 返回新的属性描述符

```typescript
function Wrap(target: any, propertyKey: string, descriptor: PropertyDescriptor): PropertyDescriptor {
  return {
    ...descriptor,
    value: function (...args: any[]) {
      console.log('包装后的方法');
      return descriptor.value.apply(this, args);
    }
  };
}
```

## 四、方法装饰器工厂

```typescript
// 带参数的日志装饰器
function LogWithPrefix(prefix: string) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;

    descriptor.value = function (...args: any[]) {
      console.log(`[${prefix}] 调用 ${propertyKey}`);
      const result = originalMethod.apply(this, args);
      console.log(`[${prefix}] ${propertyKey} 返回:`, result);
      return result;
    };
  };
}

class ApiService {
  @LogWithPrefix('API')
  fetchData(url: string) {
    return `获取 ${url} 的数据`;
  }
}
```

## 五、常见应用场景

### 1. 日志记录

```typescript
function Logger(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const startTime = Date.now();
    console.log(`→ 调用 ${propertyKey}，参数:`, args);

    const result = original.apply(this, args);

    const duration = Date.now() - startTime;
    console.log(`← ${propertyKey} 返回: ${result}，耗时: ${duration}ms`);
    return result;
  };
}

class OrderService {
  @Logger
  createOrder(items: string[]) {
    return `创建订单: ${items.join(', ')}`;
  }
}
```

### 2. 性能监控

```typescript
function PerformanceMonitor(threshold: number = 100) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = function (...args: any[]) {
      const start = performance.now();
      const result = original.apply(this, args);
      const duration = performance.now() - start;

      if (duration > threshold) {
        console.warn(`⚠️ ${propertyKey} 执行缓慢: ${duration.toFixed(2)}ms`);
      }
      return result;
    };
  };
}

class DataService {
  @PerformanceMonitor(50)
  heavyComputation(data: number[]) {
    // 模拟耗时操作
    return data.reduce((sum, n) => sum + n, 0);
  }
}
```

### 3. 防抖（Debounce）

```typescript
function Debounce(delay: number) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;
    let timer: ReturnType<typeof setTimeout> | null = null;

    descriptor.value = function (...args: any[]) {
      if (timer) clearTimeout(timer);
      timer = setTimeout(() => {
        original.apply(this, args);
      }, delay);
    };
  };
}

class SearchComponent {
  @Debounce(300)
  onSearchInput(query: string) {
    console.log(`搜索: ${query}`);
  }
}
```

### 4. 节流（Throttle）

```typescript
function Throttle(interval: number) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;
    let lastTime = 0;

    descriptor.value = function (...args: any[]) {
      const now = Date.now();
      if (now - lastTime >= interval) {
        lastTime = now;
        return original.apply(this, args);
      }
    };
  };
}

class ScrollHandler {
  @Throttle(200)
  onScroll(position: number) {
    console.log(`滚动位置: ${position}`);
  }
}
```

### 5. 缓存（Memoize）

```typescript
function Memoize(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  const cache = new Map<string, any>();

  descriptor.value = function (...args: any[]) {
    const cacheKey = JSON.stringify(args);

    if (cache.has(cacheKey)) {
      console.log(`缓存命中: ${propertyKey}(${cacheKey})`);
      return cache.get(cacheKey);
    }

    const result = original.apply(this, args);
    cache.set(cacheKey, result);
    return result;
  };
}

class MathService {
  @Memoize
  fibonacci(n: number): number {
    if (n <= 1) return n;
    return this.fibonacci(n - 1) + this.fibonacci(n - 2);
  }
}

const math = new MathService();
console.log(math.fibonacci(10)); // 计算
console.log(math.fibonacci(10)); // 命中缓存
```

### 6. 权限控制

```typescript
function RequireAuth(role: string) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = function (...args: any[]) {
      const currentUser = getCurrentUser(); // 假设的函数
      if (!currentUser || currentUser.role !== role) {
        throw new Error(`权限不足：需要 ${role} 角色`);
      }
      return original.apply(this, args);
    };
  };
}

class AdminService {
  @RequireAuth('admin')
  deleteUser(userId: string) {
    return `删除用户 ${userId}`;
  }

  @RequireAuth('admin')
  updateConfig(config: object) {
    return '更新配置';
  }
}
```

### 7. 重试机制

```typescript
function Retry(maxAttempts: number, delay: number = 1000) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;

    descriptor.value = async function (...args: any[]) {
      let lastError: Error | null = null;

      for (let attempt = 1; attempt <= maxAttempts; attempt++) {
        try {
          return await original.apply(this, args);
        } catch (error) {
          lastError = error as Error;
          console.warn(`${propertyKey} 第 ${attempt} 次尝试失败`);
          if (attempt < maxAttempts) {
            await new Promise(resolve => setTimeout(resolve, delay));
          }
        }
      }

      throw lastError;
    };
  };
}

class ApiClient {
  @Retry(3, 2000)
  async fetchData(url: string) {
    const response = await fetch(url);
    return response.json();
  }
}
```

### 8. 参数验证

```typescript
function ValidateParams(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const original = descriptor.value;
  const validators: { index: number; validator: Function }[] =
    Reflect.getMetadata('validators', target, propertyKey) || [];

  descriptor.value = function (...args: any[]) {
    for (const { index, validator } of validators) {
      if (!validator(args[index])) {
        throw new Error(
          `参数验证失败: ${propertyKey} 的第 ${index} 个参数无效`
        );
      }
    }
    return original.apply(this, args);
  };
}

class UserService {
  @ValidateParams
  createUser(name: string, age: number) {
    return { name, age };
  }
}
```

## 六、注意事项

### 1. this 指向问题

在方法装饰器中，使用普通函数（而非箭头函数）保持正确的 `this` 指向：

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  // ✅ 正确：使用普通函数，this 指向实例
  descriptor.value = function (...args: any[]) {
    console.log('this:', this); // 指向类的实例
    return original.apply(this, args);
  };

  // ❌ 错误：箭头函数会丢失 this 上下文
  // descriptor.value = (...args: any[]) => {
  //   return original.apply(this, args); // this 不正确
  // };
}
```

### 2. 静态方法 vs 实例方法

```typescript
class Example {
  @Log
  instanceMethod() {} // target 是 Example.prototype

  @Log
  static staticMethod() {} // target 是 Example（构造函数）
}
```

### 3. 保留方法名称

装饰器替换方法后，建议保留原始方法名称以便调试：

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    return original.apply(this, args);
  };

  // 保留原始方法名
  Object.defineProperty(descriptor.value, 'name', {
    value: original.name,
    configurable: true
  });
}
```
