# 参数装饰器（Parameter Decorator）

## 一、定义

**参数装饰器**是应用于类方法参数的装饰器，在参数声明之前声明。参数装饰器主要用于**标记参数**或**存储参数的元数据**，通常需要配合方法装饰器一起使用。

## 二、基本语法

```typescript
/**
 * 参数装饰器参数：
 * @param target - 构造函数（静态方法）或原型（实例方法）
 * @param propertyKey - 方法名称（undefined 对于构造函数参数）
 * @param parameterIndex - 参数在参数列表中的索引（从 0 开始）
 */
function ParameterDecorator(
  target: any,
  propertyKey: string | symbol,
  parameterIndex: number
) {
  console.log(`参数装饰器被调用`);
  console.log('target:', target);
  console.log('methodName:', propertyKey);
  console.log('parameterIndex:', parameterIndex);
}

class UserService {
  createUser(@ParameterDecorator name: string, @ParameterDecorator age: number) {
    console.log(`创建用户：${name}, ${age}岁`);
  }
}
```

### 参数说明

| 参数 | 类型 | 说明 |
|-----|------|------|
| `target` | `any` | 实例方法 → 类的原型对象；静态方法 → 类的构造函数 |
| `propertyKey` | `string \| symbol \| undefined` | 方法名称；构造函数参数时为 `undefined` |
| `parameterIndex` | `number` | 参数在参数列表中的索引（从 0 开始） |

## 三、参数装饰器的返回值

参数装饰器**返回值会被忽略**。它的作用仅限于：
1. 标记参数（存储元数据）
2. 配合方法装饰器实现验证、注入等功能

## 四、常见应用场景

### 1. 参数验证

```typescript
// 存储验证器的容器
const validationMetadata = new WeakMap<object, Map<string, { index: number; validator: Function }[]>>();

function Required(target: any, propertyKey: string, parameterIndex: number) {
  // 获取或创建验证器列表
  let validators = validationMetadata.get(target);
  if (!validators) {
    validators = new Map();
    validationMetadata.set(target, validators);
  }

  // 添加验证器
  let methodValidators = validators.get(propertyKey);
  if (!methodValidators) {
    methodValidators = [];
    validators.set(propertyKey, methodValidators);
  }

  methodValidators.push({
    index: parameterIndex,
    validator: (value: any) => value !== undefined && value !== null && value !== ''
  });
}

function Min(min: number) {
  return function (target: any, propertyKey: string, parameterIndex: number) {
    let validators = validationMetadata.get(target);
    if (!validators) {
      validators = new Map();
      validationMetadata.set(target, validators);
    }

    let methodValidators = validators.get(propertyKey);
    if (!methodValidators) {
      methodValidators = [];
      validators.set(propertyKey, methodValidators);
    }

    methodValidators.push({
      index: parameterIndex,
      validator: (value: number) => typeof value === 'number' && value >= min
    });
  };
}

// 配合方法装饰器执行验证
function ValidateParams(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const validators = validationMetadata.get(target);
    if (validators) {
      const methodValidators = validators.get(propertyKey);
      if (methodValidators) {
        for (const { index, validator } of methodValidators) {
          if (!validator(args[index])) {
            throw new Error(
              `参数验证失败: ${String(propertyKey)} 的第 ${index} 个参数无效`
            );
          }
        }
      }
    }
    return original.apply(this, args);
  };
}

class UserService {
  @ValidateParams
  createUser(@Required name: string, @Required @Min(0) age: number) {
    return { name, age };
  }
}

const service = new UserService();
console.log(service.createUser('Alice', 25)); // { name: 'Alice', age: 25 }
// service.createUser('', 25); // 报错：参数验证失败
// service.createUser('Alice', -1); // 报错：参数验证失败
```

### 2. 依赖注入

```typescript
// 服务容器
const Container = new Map<string, any>();

// 存储注入元数据
const injectMetadata = new WeakMap<object, Map<string, { index: number; token: string }[]>>();

function Inject(token: string) {
  return function (target: any, propertyKey: string, parameterIndex: number) {
    let injects = injectMetadata.get(target);
    if (!injects) {
      injects = new Map();
      injectMetadata.set(target, injects);
    }

    let methodInjects = injects.get(propertyKey);
    if (!methodInjects) {
      methodInjects = [];
      injects.set(propertyKey, methodInjects);
    }

    methodInjects.push({ index: parameterIndex, token });
  };
}

// 配合方法装饰器执行注入
function ResolveParams(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const injects = injectMetadata.get(target);
    if (injects) {
      const methodInjects = injects.get(propertyKey);
      if (methodInjects) {
        for (const { index, token } of methodInjects) {
          const service = Container.get(token);
          if (service) {
            args[index] = typeof service === 'function' ? new service() : service;
          }
        }
      }
    }
    return original.apply(this, args);
  };
}

// 注册服务
Container.set('logger', { log: (msg: string) => console.log(msg) });
Container.set('config', { apiUrl: 'https://api.example.com' });

class UserController {
  @ResolveParams
  listUsers(@Inject('logger') logger: any, @Inject('config') config: any) {
    logger.log(`从 ${config.apiUrl} 获取用户列表`);
  }
}

const controller = new UserController();
controller.listUsers(undefined as any, undefined as any);
// 输出：从 https://api.example.com 获取用户列表
```

### 3. 参数类型标记

```typescript
// 存储参数类型信息
const paramTypesMetadata = new WeakMap<object, Map<string, { index: number; type: string }[]>>();

function Typed(type: string) {
  return function (target: any, propertyKey: string, parameterIndex: number) {
    let types = paramTypesMetadata.get(target);
    if (!types) {
      types = new Map();
      paramTypesMetadata.set(target, types);
    }

    let methodTypes = types.get(propertyKey);
    if (!methodTypes) {
      methodTypes = [];
      types.set(propertyKey, methodTypes);
    }

    methodTypes.push({ index: parameterIndex, type });
  };
}

class ApiEndpoint {
  @LogParams
  createPost(
    @Typed('string') title: string,
    @Typed('string') content: string,
    @Typed('number') authorId: number
  ) {
    return { title, content, authorId };
  }
}

function LogParams(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const types = paramTypesMetadata.get(target);
    if (types) {
      const methodTypes = types.get(propertyKey);
      if (methodTypes) {
        console.log(`方法 ${String(propertyKey)} 参数类型信息:`);
        for (const { index, type } of methodTypes) {
          console.log(`  参数 ${index} (${type}):`, args[index]);
        }
      }
    }
    return original.apply(this, args);
  };
}

const api = new ApiEndpoint();
api.createPost('Hello', 'World', 1);
// 输出：
// 方法 createPost 参数类型信息:
//   参数 0 (string): Hello
//   参数 1 (string): World
//   参数 2 (number): 1
```

### 4. 路由参数绑定

```typescript
// 模拟路由参数
const routeMetadata = new WeakMap<object, Map<string, { index: number; source: string; key?: string }[]>>();

function Body(target: any, propertyKey: string, parameterIndex: number) {
  addRouteMeta(target, propertyKey, parameterIndex, 'body');
}

function Param(key: string) {
  return function (target: any, propertyKey: string, parameterIndex: number) {
    addRouteMeta(target, propertyKey, parameterIndex, 'param', key);
  };
}

function Query(key: string) {
  return function (target: any, propertyKey: string, parameterIndex: number) {
    addRouteMeta(target, propertyKey, parameterIndex, 'query', key);
  };
}

function addRouteMeta(
  target: any,
  propertyKey: string,
  parameterIndex: number,
  source: string,
  key?: string
) {
  let metas = routeMetadata.get(target);
  if (!metas) {
    metas = new Map();
    routeMetadata.set(target, metas);
  }

  let methodMetas = metas.get(propertyKey);
  if (!methodMetas) {
    methodMetas = [];
    metas.set(propertyKey, methodMetas);
  }

  methodMetas.push({ index: parameterIndex, source, key });
}

class UserController {
  getUser(@Param('id') id: string) {
    return `获取用户 ${id}`;
  }

  createUser(@Body userData: any) {
    return `创建用户 ${userData.name}`;
  }

  listUsers(@Query('page') page: number, @Query('limit') limit: number) {
    return `获取第 ${page} 页，每页 ${limit} 条`;
  }
}

// 读取路由元数据
const metas = routeMetadata.get(UserController.prototype);
metas?.forEach((value, key) => {
  console.log(`方法 ${String(key)}:`);
  value.forEach(meta => {
    console.log(`  参数 ${meta.index}: ${meta.source}${meta.key ? ` (${meta.key})` : ''}`);
  });
});
```

### 5. 参数必填标记

```typescript
const requiredParams = new WeakMap<object, Map<string, number[]>>();

function RequiredParam(target: any, propertyKey: string | symbol, parameterIndex: number) {
  let params = requiredParams.get(target);
  if (!params) {
    params = new Map();
    requiredParams.set(target, params);
  }

  let methodParams = params.get(propertyKey);
  if (!methodParams) {
    methodParams = [];
    params.set(propertyKey, methodParams);
  }

  methodParams.push(parameterIndex);
}

// 配合方法装饰器检查必填参数
function CheckRequired(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    const params = requiredParams.get(target);
    if (params) {
      const methodParams = params.get(propertyKey);
      if (methodParams) {
        for (const index of methodParams) {
          if (args[index] === undefined || args[index] === null) {
            throw new Error(
              `参数缺失: ${String(propertyKey)} 的第 ${index} 个参数是必填的`
            );
          }
        }
      }
    }
    return original.apply(this, args);
  };
}

class OrderService {
  @CheckRequired
  createOrder(
    @RequiredParam productId: string,
    quantity: number,
    @RequiredParam userId: string
  ) {
    return { productId, quantity, userId };
  }
}

const service = new OrderService();
console.log(service.createOrder('P001', 2, 'U001')); // OK
// service.createOrder('', 2, 'U001'); // 报错：参数缺失
```

## 五、参数装饰器 vs 方法装饰器

参数装饰器通常需要与方法装饰器**配合使用**：

| 特性 | 参数装饰器 | 方法装饰器 |
|-----|-----------|-----------|
| 作用目标 | 单个参数 | 整个方法 |
| 参数个数 | 3 个（target, propertyKey, parameterIndex） | 3 个（target, propertyKey, descriptor） |
| 返回值 | 被忽略 | 可选（可返回新 descriptor） |
| 主要作用 | 标记/存储元数据 | 修改方法行为 |

```typescript
// 参数装饰器负责标记
function Mark(target: any, key: string, index: number) {
  // 存储标记信息
}

// 方法装饰器负责处理
function Process(target: any, key: string, descriptor: PropertyDescriptor) {
  // 读取标记信息并执行逻辑
}

class Example {
  @Process // 方法装饰器处理
  method(@Mark param: string) {} // 参数装饰器标记
}
```

## 六、与 reflect-metadata 配合使用

参数装饰器常与 `reflect-metadata` 库配合，提供更强大的元数据反射能力：

```typescript
import 'reflect-metadata';

// 存储参数类型
function BodyParam(target: any, propertyKey: string, parameterIndex: number) {
  const existingParams: number[] =
    Reflect.getOwnMetadata('bodyParams', target, propertyKey) || [];
  existingParams.push(parameterIndex);
  Reflect.defineMetadata('bodyParams', existingParams, target, propertyKey);
}

// 读取参数类型信息
function GetParamTypes(target: any, propertyKey: string) {
  return Reflect.getMetadata('design:paramtypes', target, propertyKey);
}

class UserController {
  // reflect-metadata 会自动存储参数类型
  createUser(@BodyParam name: string, age: number) {}
}

// 获取参数类型
const types = GetParamTypes(UserController.prototype, 'createUser');
console.log(types); // [String, Number]
```

## 七、注意事项

1. **返回值被忽略**：参数装饰器的返回值不会产生任何效果
2. **必须配合其他装饰器**：单独的参数装饰器无法实现完整功能
3. **执行顺序**：参数装饰器在方法装饰器之前执行
4. **元数据存储**：需要将元数据存储在外部容器或使用 `reflect-metadata`
