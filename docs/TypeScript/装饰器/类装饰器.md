# 类装饰器（Class Decorator）

## 一、定义

**类装饰器**是应用于类构造函数的装饰器，在类声明之前声明，用来**观察、修改或替换**类的定义。

类装饰器的本质是一个函数，接收**一个参数**——类的构造函数。

## 二、基本语法

```typescript
// 类装饰器定义
function ClassDecorator(constructor: Function) {
  console.log('类装饰器被调用');
  // 可以修改类原型或类本身
  constructor.prototype.newMethod = function () {
    console.log('添加的新方法');
  };
}

@ClassDecorator
class MyClass {
  constructor() {
    console.log('类被实例化');
  }
}

const obj = new MyClass();
obj.newMethod(); // 添加的新方法
```

### 参数说明

| 参数 | 类型 | 说明 |
|-----|------|------|
| `constructor` | `Function` | 被装饰的类的构造函数 |

## 三、类装饰器的返回值

类装饰器可以有以下三种返回值：

### 1. 无返回值（void）

装饰器直接修改类的原型或静态成员：

```typescript
function AddProperty(constructor: Function) {
  constructor.prototype.createdAt = new Date();
  constructor.prototype.getType = function () {
    return this.constructor.name;
  };
}

@AddProperty
class User {
  name: string;
  constructor(name: string) {
    this.name = name;
  }
}

const user = new User('Alice');
console.log(user.getType()); // "User"
console.log(user.createdAt); // Date 对象
```

### 2. 返回新的构造函数

装饰器可以返回一个新的构造函数来**替换**原有类：

```typescript
function WithTimestamp<T extends { new (...args: any[]): {} }>(constructor: T) {
  return class extends constructor {
    timestamp = Date.now();

    getTimestamp() {
      return new Date(this.timestamp).toISOString();
    }
  };
}

@WithTimestamp
class Document {
  title: string;
  constructor(title: string) {
    this.title = title;
  }
}

const doc = new Document('TypeScript 指南');
console.log(doc.title); // "TypeScript 指南"
console.log(doc.getTimestamp()); // ISO 时间戳
```

### 3. 返回新的类（完整替换）

```typescript
function Component(constructor: Function) {
  class ExtendedComponent extends constructor {
    isComponent = true;

    render() {
      return `渲染组件: ${constructor.name}`;
    }
  }
  return ExtendedComponent;
}

@Component
class Button {
  label: string = '点击';
}

const button = new Button();
console.log(button.isComponent); // true
console.log(button.render()); // "渲染组件: Button"
```

## 四、类装饰器工厂

当需要传递参数时，使用装饰器工厂模式：

```typescript
// 类装饰器工厂
function ComponentConfig(config: { selector: string; template: string }) {
  return function (constructor: Function) {
    constructor.prototype.selector = config.selector;
    constructor.prototype.template = config.template;
    constructor.prototype.render = function () {
      return `渲染 ${config.selector}: ${config.template}`;
    };
  };
}

@ComponentConfig({
  selector: '#app',
  template: '<div>Hello World</div>'
})
class AppComponent {}

const app = new AppComponent();
console.log(app.selector); // "#app"
console.log(app.render()); // "渲染 #app: <div>Hello World</div>"
```

## 五、类型安全的类装饰器

使用泛型保持类型安全：

```typescript
// 构造函数类型
type Constructor<T = {}> = new (...args: any[]) => T;

// 为类添加单例能力的装饰器
function Singleton<T extends Constructor>(constructor: T) {
  let instance: T | null = null;

  return class extends constructor {
    static getInstance(...args: any[]): T {
      if (!instance) {
        instance = new constructor(...args);
      }
      return instance;
    }
  };
}

@Singleton
class Database {
  connection: string;
  constructor(connection: string) {
    this.connection = connection;
  }

  query(sql: string) {
    return `执行: ${sql}`;
  }
}

// 使用方式
const db1 = Database.getInstance('mysql://localhost');
const db2 = Database.getInstance('mysql://localhost');
console.log(db1 === db2); // true（同一实例）
```

## 六、常见应用场景

### 1. 依赖注入注册

```typescript
const Container = new Map<string, Function>();

function Injectable(token: string) {
  return function (constructor: Function) {
    Container.set(token, constructor);
    console.log(`服务 ${token} 已注册到容器`);
  };
}

@Injectable('userService')
class UserService {
  getUsers() {
    return ['Alice', 'Bob'];
  }
}

@Injectable('authService')
class AuthService {
  login() {
    return '登录成功';
  }
}
```

### 2. 混入（Mixin）模式

```typescript
// 可序列化 mixin
function Serializable<T extends Constructor>(constructor: T) {
  return class extends constructor {
    serialize() {
      return JSON.stringify(this);
    }

    deserialize(json: string) {
      Object.assign(this, JSON.parse(json));
      return this;
    }
  };
}

// 可观察 mixin
function Observable<T extends Constructor>(constructor: T) {
  return class extends constructor {
    private listeners: Function[] = [];

    on(event: string, callback: Function) {
      this.listeners.push(callback);
    }

    emit(event: string, data?: any) {
      this.listeners.forEach(fn => fn(data));
    }
  };
}

@Serializable
@Observable
class Model {
  data: any;
}
```

### 3. 标记与元数据

```typescript
function Entity(tableName: string) {
  return function (constructor: Function) {
    constructor.prototype._tableName = tableName;
    constructor.prototype._entityName = constructor.name;
  };
}

@Entity('users')
class User {
  id!: number;
  name!: string;
}

@Entity('orders')
class Order {
  id!: number;
  amount!: number;
}
```

## 七、注意事项

1. **返回值类型**：类装饰器返回的类型会被视为新的类定义，可能影响类型推断
2. **类型声明合并**：当装饰器添加新方法时，需要使用 `declare interface` 声明类型

```typescript
function AddMethod<T extends { new (...args: any[]): {} }>(constructor: T) {
  return class extends constructor {
    newMethod() {
      console.log('新方法');
    }
  };
}

@AddMethod
class MyClass {
  existingMethod() {}
}

// 需要声明合并以支持类型检查
declare interface MyClass {
  newMethod(): void;
}

const obj = new MyClass();
obj.newMethod(); // OK
```

3. **执行时机**：类装饰器在类定义时执行，而非实例化时
