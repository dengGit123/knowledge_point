# 访问符装饰器（Accessor Decorator）

## 一、定义

**访问符装饰器**是应用于类的 `getter` 或 `setter` 的装饰器，在访问符声明之前声明。访问符装饰器可以用来**观察、修改或替换**访问符的定义。

访问符装饰器与方法装饰器类似，都有三个参数，包括属性描述符。

## 二、基本语法

```typescript
/**
 * 访问符装饰器参数：
 * @param target - 对于静态访问符是类的构造函数，对于实例访问符是类的原型
 * @param propertyKey - 访问符名称
 * @param descriptor - 属性描述符（包含 get 和 set）
 */
function AccessorDecorator(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  console.log(`访问符装饰器被调用: ${propertyKey}`);
  console.log('descriptor:', descriptor);
}

class Person {
  private _name: string;

  constructor(name: string) {
    this._name = name;
  }

  @AccessorDecorator
  get name() {
    return this._name;
  }

  set name(value: string) {
    this._name = value;
  }
}
```

### 参数说明

| 参数 | 类型 | 说明 |
|-----|------|------|
| `target` | `any` | 实例访问符 → 类的原型对象；静态访问符 → 类的构造函数 |
| `propertyKey` | `string \| symbol` | 被装饰的访问符名称 |
| `descriptor` | `PropertyDescriptor` | 属性描述符，包含 `get` 和 `set` 方法 |

## 三、访问符装饰器的返回值

访问符装饰器可以有以下三种返回值：

### 1. 无返回值（void）

直接修改 descriptor 对象：

```typescript
function Enumerable(enumerable: boolean) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ) {
    descriptor.enumerable = enumerable;
  };
}

class Person {
  private _name: string;

  constructor(name: string) {
    this._name = name;
  }

  @Enumerable(false)
  get name() {
    return this._name;
  }
}

const person = new Person('Alice');
// name 属性不会被 for...in 或 Object.keys 枚举
console.log(Object.keys(person)); // []
```

### 2. 返回新的属性描述符

```typescript
function Configurable(configurable: boolean) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ): PropertyDescriptor {
    return {
      ...descriptor,
      configurable
    };
  };
}
```

## 四、常见应用场景

### 1. 控制可枚举性

```typescript
function Hidden(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  descriptor.enumerable = false;
}

class User {
  private _password: string;

  constructor(password: string) {
    this._password = password;
  }

  @Hidden
  get password() {
    return '******'; // 隐藏真实密码
  }
}

const user = new User('secret123');
console.log(user.password); // "******"
console.log(Object.keys(user)); // password 不会出现在键列表中
```

### 2. 缓存计算结果

```typescript
function Cached(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalGetter = descriptor.get;
  const cacheSymbol = Symbol(`_${propertyKey}_cache`);

  if (originalGetter) {
    descriptor.get = function () {
      if (!this[cacheSymbol]) {
        this[cacheSymbol] = originalGetter.call(this);
      }
      return this[cacheSymbol];
    };
  }
}

class DataProcessor {
  private data: number[];

  constructor(data: number[]) {
    this.data = data;
  }

  @Cached
  get processedData() {
    console.log('执行复杂计算...');
    return this.data.map(x => x * 2).filter(x => x > 10);
  }
}

const processor = new DataProcessor([1, 5, 10, 15, 20]);
console.log(processor.processedData); // 执行复杂计算... [20, 30, 40]
console.log(processor.processedData); // [20, 30, 40]（命中缓存）
```

### 3. 日志追踪

```typescript
function LogAccess(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalGetter = descriptor.get;
  const originalSetter = descriptor.set;

  if (originalGetter) {
    descriptor.get = function () {
      console.log(`[GET] ${propertyKey}`);
      return originalGetter.call(this);
    };
  }

  if (originalSetter) {
    descriptor.set = function (value: any) {
      console.log(`[SET] ${propertyKey} = ${value}`);
      originalSetter.call(this, value);
    };
  }
}

class State {
  private _count: number = 0;

  @LogAccess
  get count() {
    return this._count;
  }

  set count(value: number) {
    this._count = value;
  }
}

const state = new State();
console.log(state.count); // [GET] count → 0
state.count = 5; // [SET] count = 5
```

### 4. 验证赋值

```typescript
function ValidateRange(min: number, max: number) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ) {
    const originalSetter = descriptor.set;

    if (originalSetter) {
      descriptor.set = function (value: number) {
        if (value < min || value > max) {
          throw new Error(
            `${propertyKey} 必须在 ${min} 到 ${max} 之间，当前值: ${value}`
          );
        }
        originalSetter.call(this, value);
      };
    }
  };
}

class Temperature {
  private _celsius: number = 0;

  @ValidateRange(-273.15, 1000)
  get celsius() {
    return this._celsius;
  }

  set celsius(value: number) {
    this._celsius = value;
  }
}

const temp = new Temperature();
temp.celsius = 25; // OK
// temp.celsius = -300; // 报错：超出范围
```

### 5. 防抖 getter

```typescript
function DebounceGet(delay: number) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ) {
    const originalGetter = descriptor.get;
    let timer: ReturnType<typeof setTimeout> | null = null;
    let pending = false;

    if (originalGetter) {
      descriptor.get = function () {
        if (!pending) {
          pending = true;
          timer = setTimeout(() => {
            pending = false;
          }, delay);
          return originalGetter.call(this);
        }
        console.log(`[防抖] ${propertyKey} 调用过于频繁`);
        return undefined;
      };
    }
  };
}
```

### 6. 计算属性依赖追踪

```typescript
function DependsOn(...dependencies: string[]) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ) {
    const originalGetter = descriptor.get;

    if (originalGetter) {
      descriptor.get = function () {
        console.log(`[依赖] ${propertyKey} 依赖于: ${dependencies.join(', ')}`);
        return originalGetter.call(this);
      };
    }
  };
}

class ShoppingCart {
  private items: { name: string; price: number; quantity: number }[] = [];

  @DependsOn('items')
  get totalQuantity() {
    return this.items.reduce((sum, item) => sum + item.quantity, 0);
  }

  @DependsOn('items')
  get totalPrice() {
    return this.items.reduce((sum, item) => sum + item.price * item.quantity, 0);
  }
}
```

## 五、访问符装饰器 vs 方法装饰器

两者参数相同（三个参数），但作用目标不同：

| 特性 | 访问符装饰器 | 方法装饰器 |
|-----|-------------|-----------|
| 作用目标 | getter / setter | 普通方法 |
| descriptor.value | `undefined` | 方法函数 |
| descriptor.get | getter 函数 | `undefined` |
| descriptor.set | setter 函数 | `undefined` |

```typescript
// 访问符装饰器
function OnAccessor(target: any, key: string, descriptor: PropertyDescriptor) {
  console.log('get:', descriptor.get); // 有值
  console.log('set:', descriptor.set); // 有值
  console.log('value:', descriptor.value); // undefined
}

// 方法装饰器
function OnMethod(target: any, key: string, descriptor: PropertyDescriptor) {
  console.log('value:', descriptor.value); // 有值（函数）
}

class Example {
  @OnAccessor
  get value() { return 1; }

  @OnMethod
  method() { return 1; }
}
```

## 六、注意事项

### 1. 只对第一个访问符生效

当同时有 getter 和 setter 时，装饰器只应用于第一个声明的访问符：

```typescript
class Example {
  // 装饰器只应用于 getter
  @Decorator
  get value() { return this._value; }

  set value(v: any) { this._value = v; } // 没有装饰器

  // 装饰器只应用于 setter
  get other() { return this._other; }

  @Decorator
  set other(v: any) { this._other = v; }
}
```

### 2. 共享同一个 descriptor

同一个属性的 getter 和 setter **共享同一个属性描述符**：

```typescript
function LogBoth(target: any, key: string, descriptor: PropertyDescriptor) {
  // 修改 getter 会影响 setter，因为它们共享 descriptor
  const originalGet = descriptor.get;
  descriptor.get = function () {
    console.log(`读取 ${key}`);
    return originalGet?.call(this);
  };
}

class Example {
  @LogBoth
  get value() { return 1; } // 装饰器应用在这里

  set value(v: number) { } // 但由于共享 descriptor，setter 也会受影响
}
```

### 3. 类型安全

```typescript
function TypedAccessor(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalGet = descriptor.get;
  const originalSet = descriptor.set;

  if (originalGet) {
    descriptor.get = function (): unknown {
      return originalGet.call(this);
    };
  }

  if (originalSet) {
    descriptor.set = function (value: unknown) {
      originalSet.call(this, value);
    };
  }
}
```
