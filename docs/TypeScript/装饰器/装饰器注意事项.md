# 装饰器注意事项与最佳实践

## 一、TypeScript 版本差异

### 1. 实验性装饰器 vs 新装饰器标准

TypeScript 中存在两套装饰器标准，它们**不兼容**：

| 特性 | 实验性装饰器 | 新装饰器标准 (Stage 3) |
|-----|-------------|----------------------|
| 引入时间 | TypeScript 1.5+ | TypeScript 5.0+ |
| 配置项 | `experimentalDecorators` | 无需特殊配置 |
| 标准状态 | 旧提案 | ECMAScript Stage 3 |
| 元数据支持 | `emitDecoratorMetadata` | 通过 `Symbol.metadata` |
| 参数装饰器 | 支持 | 支持（签名不同） |

### 2. 配置方式

```json
// tsconfig.json

// 实验性装饰器（旧版）
{
  "compilerOptions": {
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}

// 新装饰器标准（TypeScript 5.0+）
{
  "compilerOptions": {
    "target": "ES2022",
    "experimentalDecorators": false,  // 关闭旧版
    // 新标准无需额外配置
  }
}
```

### 3. 两种标准的区别

```typescript
// ===== 实验性装饰器（旧版） =====
function OldDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  // descriptor 是 PropertyDescriptor
}

// ===== 新装饰器标准（Stage 3） =====
function NewDecorator(context: ClassMethodDecoratorContext) {
  // 接收 context 对象，返回替换函数
  return function (this: any, ...args: any[]) {
    return /* 原始方法的返回值 */;
  };
}
```

> **建议**：新项目推荐使用新标准，旧项目保持实验性装饰器以确保兼容性。

## 二、类型安全问题

### 1. 装饰器破坏类型推断

装饰器可能破坏 TypeScript 的类型推断：

```typescript
function AddMethod<T extends { new (...args: any[]): {} }>(constructor: T) {
  return class extends constructor {
    newMethod() {
      console.log('新方法');
    }
  };
}

@AddMethod
class MyClass {
  existingMethod() {}
}

const obj = new MyClass();
obj.existingMethod(); // OK
obj.newMethod(); // ❌ TypeScript 不知道这个方法
```

### 2. 解决方案：声明合并

```typescript
// 使用声明合并添加类型
declare interface MyClass {
  newMethod(): void;
}

const obj = new MyClass();
obj.newMethod(); // ✅ OK
```

### 3. 使用泛型保持类型

```typescript
// 更好的方式：使用泛型约束
function AddMethod<T extends { new (...args: any[]): InstanceType<T> }>(
  constructor: T
) {
  return class extends constructor {
    newMethod() {
      console.log('新方法');
    }
  };
}
```

### 4. 方法装饰器的类型安全

```typescript
// 类型安全的方法装饰器
function Log<T>(
  target: T,
  propertyKey: string,
  descriptor: TypedPropertyDescriptor<(...args: any[]) => any>
) {
  const original = descriptor.value!;

  descriptor.value = function (this: T, ...args: any[]) {
    console.log(`调用 ${propertyKey}`);
    return original.apply(this, args);
  };
}
```

## 三、this 指向问题

### 1. 方法装饰器中的 this

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  // ✅ 正确：使用普通函数，this 指向实例
  descriptor.value = function (...args: any[]) {
    console.log('this:', this); // 指向类的实例
    return original.apply(this, args);
  };

  // ❌ 错误：箭头函数会丢失 this 上下文
  // descriptor.value = (...args: any[]) => {
  //   return original.apply(this, this); // this 不正确
  // };
}
```

### 2. 类装饰器中的 this

```typescript
function ClassDecorator<T extends { new (...args: any[]): {} }>(constructor: T) {
  return class extends constructor {
    constructor(...args: any[]) {
      super(...args);
      // 这里的 this 指向新类的实例
      console.log('新类实例创建');
    }
  };
}
```

## 四、性能考虑

### 1. 装饰器的执行时机

```typescript
// 装饰器在类定义时执行（编译时/模块加载时）
function HeavyDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  // 这里的代码在类定义时执行一次
  console.log('装饰器执行');

  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    // 这里的代码在每次方法调用时执行
    return original.apply(this, args);
  };
}
```

### 2. 避免在装饰器中执行重操作

```typescript
// ❌ 不好：装饰器中执行重操作
function BadDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  // 每次类定义时都会执行
  const data = fs.readFileSync('./config.json'); // 同步读取文件
  const config = JSON.parse(data.toString());
  // ...
}

// ✅ 好：延迟执行
function GoodDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  let config: any = null;

  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    // 只在首次调用时加载
    if (!config) {
      config = loadConfig();
    }
    return original.apply(this, args);
  };
}
```

### 3. 缓存装饰器的内存泄漏

```typescript
// ⚠️ 注意：缓存可能导致内存泄漏
function Memoize(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  const cache = new Map(); // 这个 Map 永远不会被垃圾回收

  descriptor.value = function (...args: any[]) {
    const cacheKey = JSON.stringify(args);
    if (cache.has(cacheKey)) {
      return cache.get(cacheKey);
    }
    const result = original.apply(this, args);
    cache.set(cacheKey, result);
    return result;
  };
}

// ✅ 改进：使用 WeakMap 或添加清理方法
function MemoizeSafe(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  const cache = new Map();

  descriptor.value = function (...args: any[]) {
    const cacheKey = JSON.stringify(args);
    if (cache.has(cacheKey)) {
      return cache.get(cacheKey);
    }
    const result = original.apply(this, args);
    cache.set(cacheKey, result);
    return result;
  };

  // 添加清理方法
  (descriptor.value as any).clearCache = () => cache.clear();
}
```

## 五、调试困难

### 1. 堆栈追踪问题

装饰器包装方法后，堆栈追踪可能变得复杂：

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    try {
      return original.apply(this, args);
    } catch (error) {
      // 添加装饰器信息到错误
      console.error(`错误发生在 ${String(key)} 方法`);
      throw error;
    }
  };
}
```

### 2. 保留方法名称

```typescript
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    return original.apply(this, args);
  };

  // 保留原始方法名
  Object.defineProperty(descriptor.value, 'name', {
    value: original.name,
    configurable: true
  });
}
```

### 3. 添加有意义的日志

```typescript
function DebugLog(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  const className = target.constructor.name;

  descriptor.value = function (...args: any[]) {
    console.log(`[${className}.${String(key)}] 参数:`, args);
    const result = original.apply(this, args);
    console.log(`[${className}.${String(key)}] 返回:`, result);
    return result;
  };
}
```

## 六、兼容性问题

### 1. 装饰器是 ECMAScript 提案

装饰器目前处于 **Stage 3** 提案阶段，尚未正式纳入 ECMAScript 标准。

### 2. 需要编译器支持

```json
// tsconfig.json
{
  "compilerOptions": {
    "experimentalDecorators": true,  // TypeScript
    "target": "ES2022"
  }
}
```

```js
// babel.config.js
// Babel 需要 @babel/plugin-proposal-decorators
{
  "plugins": [
    ["@babel/plugin-proposal-decorators", { "version": "2023-05" }]
  ]
}
```

### 3. 运行时可能需要 polyfill

```typescript
// 对于旧版装饰器，可能需要 reflect-metadata
import 'reflect-metadata';
```

## 七、最佳实践

### 1. 单一职责原则

```typescript
// ❌ 不好：一个装饰器做太多事
function DoEverything(target: any, key: string, descriptor: PropertyDescriptor) {
  // 日志 + 验证 + 缓存 + 权限检查...
}

// ✅ 好：每个装饰器只做一件事
function Log(...) { /* 只记录日志 */ }
function Validate(...) { /* 只验证参数 */ }
function Cache(...) { /* 只处理缓存 */ }
function RequireAuth(...) { /* 只检查权限 */ }

// 组合使用
class Service {
  @RequireAuth('admin')
  @Log
  @Cache
  @Validate
  sensitiveOperation() {}
}
```

### 2. 可组合性

```typescript
// ✅ 好的装饰器设计：可独立使用和组合
function Debounce(delay: number) {
  return function (target: any, key: string, descriptor: PropertyDescriptor) {
    const original = descriptor.value;
    let timer: ReturnType<typeof setTimeout> | null = null;

    descriptor.value = function (...args: any[]) {
      if (timer) clearTimeout(timer);
      timer = setTimeout(() => original.apply(this, args), delay);
    };
  };
}

function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`调用 ${String(key)}`);
    return original.apply(this, args);
  };
}

// 可以任意组合
class Component {
  @Debounce(300)
  @Log
  handleInput(value: string) {
    console.log('处理输入:', value);
  }

  @Log
  @Debounce(500)
  handleScroll(position: number) {
    console.log('处理滚动:', position);
  }
}
```

### 3. 文档化

```typescript
/**
 * 方法调用日志装饰器
 * 
 * @description 记录方法的调用参数和返回值
 * @example
 * ```typescript
 * class MyService {
 *   @Log
 *   getData(id: string) {
 *     return { id, name: 'Test' };
 *   }
 * }
 * ```
 */
function Log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`→ ${String(key)}`, args);
    const result = original.apply(this, args);
    console.log(`← ${String(key)}`, result);
    return result;
  };
}
```

### 4. 类型安全

```typescript
// ✅ 提供完整的类型定义
type MethodDecorator = <T>(
  target: Object,
  propertyKey: string | symbol,
  descriptor: TypedPropertyDescriptor<T>
) => TypedPropertyDescriptor<T> | void;

function TypedLog<T extends (...args: any[]) => any>(
  target: any,
  key: string,
  descriptor: TypedPropertyDescriptor<T>
): TypedPropertyDescriptor<T> {
  const original = descriptor.value!;

  descriptor.value = function (this: any, ...args: any[]) {
    console.log(`调用 ${key}`);
    return original.apply(this, args);
  } as T;

  return descriptor;
}
```

### 5. 性能优化

```typescript
// ✅ 避免在热路径上使用复杂装饰器
function LightweightLog(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  // 只在开发环境启用
  if (process.env.NODE_ENV !== 'development') {
    return;
  }

  descriptor.value = function (...args: any[]) {
    console.log(`→ ${String(key)}`);
    return original.apply(this, args);
  };
}
```

### 6. 错误处理

```typescript
// ✅ 装饰器中添加错误处理
function SafeDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function (...args: any[]) {
    try {
      return original.apply(this, args);
    } catch (error) {
      console.error(`装饰器 ${String(key)} 执行出错:`, error);
      throw error; // 重新抛出，不吞掉错误
    }
  };
}
```

## 八、何时使用 / 何时不使用

### ✅ 适合使用装饰器的场景

- 开发框架或库
- 需要统一处理横切关注点（日志、权限、事务）
- 需要元数据驱动的开发模式
- 简化重复性代码
- 实现 AOP 编程

### ❌ 不适合使用装饰器的场景

- 简单的增删改查业务
- 性能敏感的热路径
- 团队不熟悉装饰器
- 过度设计（为用而用）
- 需要兼容旧版浏览器（无编译工具）

## 九、常见错误

### 1. 箭头函数导致 this 丢失

```typescript
// ❌ 错误
function BadDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = (...args: any[]) => {
    return original.apply(this, args); // this 不正确
  };
}

// ✅ 正确
function GoodDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (this: any, ...args: any[]) {
    return original.apply(this, args); // this 正确
  };
}
```

### 2. 忘记返回 descriptor

```typescript
// ❌ 错误：没有返回 descriptor
function BadDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    return original.apply(this, args);
  };
  // 忘记 return descriptor
}

// ✅ 正确
function GoodDecorator(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function (...args: any[]) {
    return original.apply(this, args);
  };
  return descriptor; // 显式返回
}
```

### 3. 装饰器工厂忘记调用

```typescript
// ❌ 错误：忘记调用工厂函数
class Example {
  @Log  // 错误：Log 是装饰器工厂，需要调用
  method() {}
}

// ✅ 正确
class Example {
  @Log()  // 正确：调用工厂函数
  method() {}
}
```
