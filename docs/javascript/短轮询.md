### 一、概述

**短轮询（Short Polling）** 是最简单的一种客户端与服务器通信模式，核心思路是：客户端**按照固定的时间间隔**，周期性地向服务器发请求，询问"有没有新数据"。无论服务器是否有新数据，每次都会立即响应——有数据就返回数据，没数据就返回空结果。

> 📖 相关规范：[Fetch Standard - WHATWG](https://fetch.spec.whatwg.org/) | [setInterval - MDN](https://developer.mozilla.org/zh-CN/docs/Web/API/setInterval)

### 二、基本概念与核心原理

#### 1. 用生活场景理解

想象你在等外卖：

- **短轮询**：每隔 5 分钟打一次电话问"外卖到哪了"，不管外卖有没有在送，到点就问
- **长轮询**：打一次电话后不挂，等外卖员接通告诉你"到了"才挂
- **WebSocket**：外卖员和你保持通话，随时汇报位置

短轮询的本质就是**定时轮询**——用定时器驱动，周期性发起 HTTP 请求。

#### 2. 工作流程

```
客户端                                        服务器
  |                                              |
  |------- 请求：有新数据吗？ ------------------>|
  |<------ 响应：没有（或返回数据） -------------|
  |                                              |
  |          ... 等待固定间隔（如 2 秒） ...      |
  |                                              |
  |------- 请求：有新数据吗？ ------------------>|
  |<------ 响应：有，这是新数据 -----------------|
  |                                              |
  |          ... 等待固定间隔（如 2 秒） ...      |
  |                                              |
  |                   （循环往复）                |
```

**关键特征**：

- **定时驱动**：用 `setInterval` 或递归 `setTimeout` 触发请求
- **请求即响应**：服务器收到请求后立即返回，不做任何等待
- **无状态**：每次请求都是独立的，服务器不保留客户端的状态

#### 3. 三种轮询模式对比

| 模式 | 工作方式 | 实时性 | 服务器压力 |
| --- | --- | --- | --- |
| **短轮询** | 固定间隔发请求（如每 2 秒一次） | 平均延迟 = 间隔 / 2 | 大量空请求，浪费资源 |
| **长轮询** | 服务器挂起请求直到有数据 | 接近实时 | 连接占用时间长 |
| **WebSocket** | 全双工通道，随时互发 | 真正实时 | 连接数少，需维持状态 |

### 三、详细用法

#### 1. 基本用法

##### 最简单的实现：`setInterval` + `fetch`

```javascript
/**
 * 最简单的短轮询实现
 * 每 3 秒请求一次服务器获取最新消息
 */
function startPolling() {
  // 每 3000 毫秒（3 秒）执行一次
  setInterval(async () => {
    try {
      const response = await fetch('/api/messages/latest');
      const data = await response.json();

      if (data.messages.length > 0) {
        console.log('收到新消息:', data.messages);
        renderMessages(data.messages);
      }
    } catch (error) {
      console.error('轮询请求失败:', error);
    }
  }, 3000);
}

// 页面加载后启动轮询
startPolling();
```

##### 递归 `setTimeout` 实现（更推荐）

```javascript
/**
 * 用递归 setTimeout 实现短轮询
 * 比 setInterval 更好：能保证请求间隔，避免请求堆积
 */
async function pollWithTimeout() {
  try {
    const response = await fetch('/api/notifications');
    const data = await response.json();
    renderNotifications(data.notifications);
  } catch (error) {
    console.error('轮询失败:', error);
  } finally {
    // 无论成功还是失败，都等 3 秒后再发起下一次
    // ✅ 这样能保证：上一次请求完成后才等待，间隔是"请求 + 等待"
    setTimeout(pollWithTimeout, 3000);
  }
}

// 启动轮询
pollWithTimeout();
```

> 💡 **为什么 `setTimeout` 递归比 `setInterval` 更好？**
>
> - `setInterval` 不管上一次请求是否完成，到点就发下一次——如果网络慢、请求耗时长，会导致**请求堆积**
> - `setTimeout` 递归是在上一次请求**完成之后**才等待，保证同一时刻只有一个请求在进行

```javascript
// ❌ setInterval 的问题：请求可能堆积
setInterval(async () => {
  // 假设这个请求需要 5 秒才返回
  await fetch('/api/slow-endpoint');
}, 3000);
// 结果：第一个请求还没返回，第二个、第三个请求已经发出去了

// ✅ setTimeout 递归：保证请求串行
async function poll() {
  await fetch('/api/slow-endpoint');
  setTimeout(poll, 3000); // 等请求完成后再等 3 秒
}
poll();
```

#### 2. 进阶用法

##### 带轮询控制的封装类

```javascript
/**
 * 短轮询控制器
 * 提供启动、暂停、停止等控制能力
 */
class ShortPollingController {
  constructor(url, interval = 3000, options = {}) {
    this.url = url;
    this.interval = interval;
    this.timerId = null;
    this.isRunning = false;
    this.immediate = options.immediate || false; // 是否立即执行一次
    this.onError = options.onError || (() => {});
    this.onData = options.onData || (() => {});
  }

  // 启动轮询
  start() {
    if (this.isRunning) return;
    this.isRunning = true;

    // 是否立即执行一次，不用等第一个间隔
    if (this.immediate) {
      this._poll();
    } else {
      this.timerId = setTimeout(() => this._poll(), this.interval);
    }
  }

  // 暂停轮询
  pause() {
    this.isRunning = false;
    if (this.timerId) {
      clearTimeout(this.timerId);
      this.timerId = null;
    }
  }

  // 恢复轮询
  resume() {
    if (!this.isRunning) {
      this.start();
    }
  }

  // 停止轮询（彻底停止，需要重新调用 start）
  stop() {
    this.pause();
  }

  // 修改轮询间隔
  setInterval(newInterval) {
    this.interval = newInterval;
    // 如果正在运行，重启以应用新间隔
    if (this.isRunning) {
      this.pause();
      this.start();
    }
  }

  // 核心轮询逻辑
  async _poll() {
    if (!this.isRunning) return;

    try {
      const response = await fetch(this.url);

      if (!response.ok) {
        throw new Error(`HTTP 错误: ${response.status}`);
      }

      const data = await response.json();
      this.onData(data);
    } catch (error) {
      this.onError(error);
    } finally {
      // 继续下一轮
      if (this.isRunning) {
        this.timerId = setTimeout(() => this._poll(), this.interval);
      }
    }
  }
}

// 使用示例
const poller = new ShortPollingController('/api/status', 5000, {
  immediate: true,           // 立即执行一次
  onData: (data) => {        // 收到数据的回调
    console.log('当前状态:', data.status);
    updateUI(data);
  },
  onError: (error) => {      // 出错的回调
    console.error('轮询出错:', error);
    showErrorToast('连接异常');
  }
});

poller.start();
```

##### 带指数退避的错误处理

```javascript
/**
 * 带退避策略的短轮询
 * 当请求连续失败时，自动拉长间隔，避免雪崩
 */
class BackoffPollingController {
  constructor(url, options = {}) {
    this.url = url;
    this.baseInterval = options.baseInterval || 3000;     // 基础间隔
    this.maxInterval = options.maxInterval || 60000;      // 最大间隔
    this.currentInterval = this.baseInterval;
    this.isRunning = false;
    this.consecutiveErrors = 0;                            // 连续错误次数
  }

  async start() {
    this.isRunning = true;

    while (this.isRunning) {
      try {
        const response = await fetch(this.url);
        if (!response.ok) throw new Error(`HTTP ${response.status}`);

        const data = await response.json();

        // 请求成功，重置间隔和错误计数
        this.currentInterval = this.baseInterval;
        this.consecutiveErrors = 0;

        this.onData?.(data);
      } catch (error) {
        this.consecutiveErrors++;
        this.onError?.(error, this.consecutiveErrors);

        // 指数退避：每次错误间隔翻倍，但不超过最大值
        this.currentInterval = Math.min(
          this.currentInterval * 2,
          this.maxInterval
        );
      }

      // 等待当前间隔后再发起下一次
      await this._delay(this.currentInterval);
    }
  }

  stop() {
    this.isRunning = false;
  }

  _delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// 使用：网络差时自动拉长请求间隔
const poller = new BackoffPollingController('/api/data', {
  baseInterval: 3000,
  maxInterval: 60000
});
poller.start();
```

##### 页面可见性控制：页面不可见时暂停轮询

```javascript
/**
 * 智能短轮询：页面不可见时暂停，切回来时恢复
 * 节省网络和服务器资源
 */
class SmartPollingController extends ShortPollingController {
  constructor(url, interval, options = {}) {
    super(url, interval, options);
    this._setupVisibilityHandler();
  }

  _setupVisibilityHandler() {
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        // 页面不可见：暂停轮询
        console.log('页面不可见，暂停轮询');
        this.pause();
      } else {
        // 页面可见：恢复轮询，并立即拉取一次最新数据
        console.log('页面可见，恢复轮询');
        this.resume();
        this._poll(); // 立即执行一次
      }
    });
  }
}
```

> 💡 **提示：** `visibilitychange` 事件在页面切换标签页、最小化浏览器时触发。`document.hidden` 为 `true` 表示页面当前不可见。

#### 3. 服务端实现（Node.js / Express）

短轮询的服务端实现非常简单——就是一个普通的接口，收到请求立即返回：

```javascript
const express = require('express');
const app = express();

// 模拟消息存储
let messages = [];
let lastId = 0;

// 短轮询接口：直接返回当前所有未读消息
app.get('/api/messages/latest', (req, res) => {
  // 获取客户端已读的最后消息编号
  const lastSeenId = parseInt(req.query.lastId) || 0;

  // 筛选出未读消息并立即返回
  const unreadMessages = messages.filter(m => m.id > lastSeenId);

  res.json({
    messages: unreadMessages,
    serverTime: Date.now()
  });
});

// 发送消息接口
app.post('/api/messages/send', express.json(), (req, res) => {
  const message = {
    id: ++lastId,
    content: req.body.content,
    timestamp: Date.now()
  };
  messages.push(message);

  res.json({ success: true, message });
});

app.listen(3000, () => {
  console.log('服务器运行在 http://localhost:3000');
});
```

#### 4. 多实例轮询管理

```javascript
/**
 * 轮询管理器：统一管理页面上的多个轮询任务
 * 页面卸载时一键清理，避免内存泄漏
 */
class PollingManager {
  constructor() {
    this.pollers = new Map();

    // 页面卸载前清理所有轮询
    window.addEventListener('beforeunload', () => {
      this.stopAll();
    });
  }

  // 注册并启动一个轮询
  add(name, poller) {
    // 同名任务先停掉旧的
    if (this.pollers.has(name)) {
      this.pollers.get(name).stop();
    }
    this.pollers.set(name, poller);
    poller.start();
  }

  // 移除并停止指定轮询
  remove(name) {
    if (this.pollers.has(name)) {
      this.pollers.get(name).stop();
      this.pollers.delete(name);
    }
  }

  // 停止所有轮询
  stopAll() {
    this.pollers.forEach(poller => poller.stop());
    this.pollers.clear();
  }
}

// 使用
const manager = new PollingManager();
manager.add('messages', new ShortPollingController('/api/messages', 3000));
manager.add('notifications', new ShortPollingController('/api/notifications', 10000));
manager.add('status', new ShortPollingController('/api/status', 5000));

// 切换页面时清理
// manager.stopAll();
```

### 五、常见问题与注意事项

#### 1. 轮询间隔怎么选？

| 间隔 | 适用场景 | 优缺点 |
| --- | --- | --- |
| 1~3 秒 | 实时性要求高的场景（如聊天预览） | 请求频繁，服务器压力大 |
| 5~10 秒 | 一般实时通知（如消息提醒） | 平衡实时性和压力 |
| 30~60 秒 | 实时性要求不高（如数据看板） | 服务器压力小，但延迟明显 |
| 大于 60 秒 | 低优先级数据刷新（如公告） | 接近"手动刷新"体验 |

> ⚠️ **注意：** 间隔越短，实时性越好，但服务器压力也越大。需要根据实际业务场景权衡，避免无意义的频繁请求。

#### 2. 请求堆积问题

```javascript
// ❌ 错误：用 setInterval 做异步请求，可能堆积
setInterval(async () => {
  await fetch('/api/slow-query'); // 假设需要 5 秒
}, 3000);
// 第一个请求还没返回，第二个又发出了

// ✅ 正确：用 setTimeout 递归，保证串行
async function poll() {
  await fetch('/api/slow-query');
  setTimeout(poll, 3000);
}
poll();
```

#### 3. 内存泄漏防范

```javascript
// ❌ 错误：页面跳转或组件销毁前没清理定时器
mounted() {
  this.timer = setInterval(this.fetchData, 3000);
}
// 组件销毁后定时器还在跑，导致内存泄漏

// ✅ 正确：销毁前清理
// Vue 选项式 API
mounted() {
  this.timer = setInterval(this.fetchData, 3000);
},
beforeUnmount() {
  clearInterval(this.timer);
}

// ✅ Vue 组合式 API
import { onMounted, onUnmounted } from 'vue';

onMounted(() => {
  timerId = setInterval(fetchData, 3000);
});

onUnmounted(() => {
  clearInterval(timerId);
});
```

#### 4. 网络带宽浪费

短轮询最大的问题是**大量空请求**——如果数据变化不频繁，大部分请求都返回空结果，白白消耗带宽。

**优化策略**：

- **按需轮询**：只在用户活跃时轮询，页面不可见时暂停
- **动态间隔**：数据变化频繁时缩短间隔，长时间无变化时拉大间隔
- **条件轮询**：服务端返回"下次最早检查时间"，客户端按这个时间等待

```javascript
// 动态间隔示例：根据数据更新频率自适应
class AdaptivePollingController {
  constructor(url, options = {}) {
    this.url = url;
    this.minInterval = options.minInterval || 2000;
    this.maxInterval = options.maxInterval || 30000;
    this.currentInterval = options.initialInterval || 5000;
    this.isRunning = false;
  }

  async start() {
    this.isRunning = true;

    while (this.isRunning) {
      const { data, hasChanges } = await this._poll();

      if (hasChanges) {
        // 有变化：缩短间隔（更频繁地检查）
        this.currentInterval = Math.max(
          this.currentInterval * 0.8,
          this.minInterval
        );
      } else {
        // 无变化：拉长间隔（减少请求）
        this.currentInterval = Math.min(
          this.currentInterval * 1.5,
          this.maxInterval
        );
      }
    }
  }
}
```

#### 5. 服务端压力

> ⚠️ **注意：** 短轮询对服务器不友好。1000 个用户、每 3 秒一次请求 = 服务器每秒要处理约 333 个请求，其中大部分返回空结果。

**缓解方案**：

- 使用 CDN 或缓存层（如 Redis）缓存响应
- 服务端返回 `304 Not Modified` 表示数据未变化
- 考虑升级为长轮询或 SSE

### 六、与相关概念对比

#### 1. 短轮询 vs 长轮询

| 对比维度 | 短轮询 | 长轮询 |
| --- | --- | --- |
| **请求频率** | 固定间隔，与是否有数据无关 | 收到响应后才发起下一个 |
| **实时性** | 平均延迟 = 间隔时间 / 2 | 接近实时（有数据立即响应） |
| **空请求比例** | 高（大部分请求返回空） | 低（每次返回都有意义） |
| **实现复杂度** | 非常简单 | 需要处理连接挂起和超时 |
| **服务器压力** | 请求次数多 | 连接占用时间长 |
| **兼容性** | 所有浏览器 | 所有浏览器 |

#### 2. 短轮询 vs SSE vs WebSocket

| 对比维度 | 短轮询 | SSE（EventSource） | WebSocket |
| --- | --- | --- | --- |
| **通信方向** | 客户端单向请求 | 单向：服务器→客户端 | 全双工双向通信 |
| **协议** | HTTP/1.1 | HTTP/1.1 | 独立协议 |
| **实时性** | 取决于间隔 | 接近实时 | 真正实时 |
| **连接复用** | 每次请求重新建连 | 一条长连接持续推送 | 一条连接双向复用 |
| **断线重连** | 无需处理（下次自动重试） | 浏览器自动重连 | 需手动实现 |
| **实现难度** | ⭐ 最简单 | ⭐⭐ 简单 | ⭐⭐⭐ 较复杂 |
| **适用场景** | 简单刷新、兼容性要求高 | 新闻推送、日志流 | 聊天、游戏、协同编辑 |

#### 3. 如何选择

```
需要实时双向通信？
  └─ 是 → WebSocket
  └─ 否 → 实时性要求高（秒级以下）？
            └─ 是 → 不需要兼容 IE？
            │         └─ 是 → SSE
            │         └─ 否 → 长轮询
            └─ 否 → 短轮询 ✅（最简单够用）
```

### 七、总结

**短轮询的核心要点回顾：**

- **本质**：用定时器驱动，周期性发起 HTTP 请求查询新数据
- **优点**：实现简单、无需特殊服务端支持、兼容性最好、天然支持断线重连
- **缺点**：实时性差（取决于间隔）、大量空请求浪费资源、服务器压力大
- **最佳场景**：实时性要求不高（秒级以上）、需要兼容老浏览器、快速实现原型
- **核心建议**：优先用 `setTimeout` 递归而非 `setInterval`；页面不可见时暂停轮询；组件销毁前清理定时器

> 💡 **提示：** 短轮询是"能用但不够好"的方案。在新项目中，如果浏览器环境允许，**SSE 是更好的选择**——它既保持了 HTTP 的简单性，又避免了短轮询的资源浪费。短轮询的真正价值在于**实现简单**和**兼容性最好**。

    <p class="update-time">最后更新：{{ lastUpdateTime }}</p>
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';

const sales = ref(0);
const orderCount = ref(0);
const lastUpdateTime = ref('--');

let timerId = null;

async function fetchDashboardData() {
  try {
    const response = await fetch('/api/dashboard/sales');
    const data = await response.json();

    sales.value = data.sales;
    orderCount.value = data.orderCount;
    lastUpdateTime.value = new Date().toLocaleTimeString();
  } catch (error) {
    console.error('获取数据失败:', error);
  }
}

// 启动短轮询（带智能控制）
function startPolling() {
  // 立即获取一次
  fetchDashboardData();

  // 每 10 秒轮询一次
  timerId = setInterval(fetchDashboardData, 10000);

  // 页面不可见时暂停，节省资源
  document.addEventListener('visibilitychange', handleVisibilityChange);
}

function handleVisibilityChange() {
  if (document.hidden) {
    // 页面隐藏：清除定时器
    clearInterval(timerId);
    timerId = null;
  } else if (!timerId) {
    // 页面显示：重新启动，并立即拉取最新数据
    fetchDashboardData();
    timerId = setInterval(fetchDashboardData, 10000);
  }
}

onMounted(() => {
  startPolling();
});

onUnmounted(() => {
  // 组件卸载时清理定时器，避免内存泄漏
  clearInterval(timerId);
  document.removeEventListener('visibilitychange', handleVisibilityChange);
});
</script>
