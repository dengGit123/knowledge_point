### 一、概述

**长轮询（Long Polling）** 是一种基于 HTTP 的服务器推送技术，用于实现客户端与服务器之间的**准实时通信**。它的核心思想是：客户端发出请求后，服务器**故意不立即响应**，而是把连接"挂起"，直到有新数据要推送时才返回结果——从而避免了普通轮询带来的频繁空请求问题。

> 📖 相关规范：[Fetch Standard - WHATWG](https://fetch.spec.whatwg.org/) | [XMLHttpRequest - MDN](https://developer.mozilla.org/zh-CN/docs/Web/API/XMLHttpRequest)

### 二、基本概念与核心原理

要理解长轮询，先要理解它的"前辈"——普通轮询（Polling）。

#### 1. 三种通信模式对比

| 模式 | 工作方式 | 实时性 | 服务器压力 |
| --- | --- | --- | --- |
| **短轮询（Polling）** | 客户端每隔固定时间就问一次"有新消息吗？" | 延迟等于轮询间隔 | 大量空请求，浪费资源 |
| **长轮询（Long Polling）** | 客户端问一次，服务器"憋着不答"直到有数据 | 接近实时 | 连接占用时间长 |
| **WebSocket** | 建立一条全双工通道，双方随时互发数据 | 真正实时 | 连接数少，但需维持状态 |

#### 2. 长轮询的工作流程

用生活场景类比：**短轮询**就像每隔 5 分钟打一次快递电话问"我的包裹到没到"；**长轮询**则是打一次电话后，快递员拿着电话不挂，包裹到了才告诉你。

```
客户端                                        服务器
  |                                              |
  |------- HTTP 请求：有新消息吗？ -------------->|
  |                                              |
  |                   （服务器挂起连接，          |
  |                     等待新数据到来）          |
  |                                              |
  |<------ HTTP 响应：这是你的新数据 ------------|
  |                                              |
  |------- 立即发起下一个 HTTP 请求 ----------->|
  |                   （循环往复）                |
```

**关键机制**：

- **挂起连接**：服务器收到请求后不立即返回，而是保持连接打开
- **超时控制**：连接不能无限期挂起，通常设置一个超时时间（如 30 秒）
- **断线重连**：客户端收到响应（或超时）后，立即发起下一个请求，形成循环

### 三、详细用法

#### 1. 基本用法

##### 客户端实现（原生 `fetch`）

```javascript
/**
 * 长轮询客户端
 * 核心逻辑：请求 -> 收到响应 -> 处理数据 -> 立即发起下一个请求
 */
async function longPolling() {
  try {
    // 向服务器发起长轮询请求
    const response = await fetch('/api/messages/poll', {
      method: 'GET',
      headers: {
        'Content-Type': 'application/json'
      }
    });

    if (!response.ok) {
      throw new Error(`HTTP 错误: ${response.status}`);
    }

    const data = await response.json();
    console.log('收到新消息:', data);

    // 处理接收到的数据
    handleMessages(data);
  } catch (error) {
    // 出错时等一段时间再重试，避免疯狂重连
    console.error('长轮询出错:', error);
    await delay(5000);
  } finally {
    // 无论成功还是失败，都要发起下一个请求（形成循环）
    longPolling();
  }
}

// 启动长轮询
longPolling();

// 工具函数：延迟指定毫秒
function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

// 处理消息
function handleMessages(messages) {
  messages.forEach(msg => {
    const li = document.createElement('li');
    li.textContent = msg.content;
    document.getElementById('message-list').appendChild(li);
  });
}
```

##### 带超时控制的改进版

```javascript
/**
 * 带超时和退避策略的长轮询
 * 更健壮的实现，适用于生产环境
 */
class LongPollingClient {
  constructor(url, options = {}) {
    this.url = url;
    this.timeout = options.timeout || 30000;      // 单次请求超时
    this.retryDelay = options.retryDelay || 5000;  // 重试间隔
    this.maxRetryDelay = options.maxRetryDelay || 60000; // 最大重试间隔
    this.currentRetryDelay = this.retryDelay;
    this.isRunning = false;
  }

  // 启动长轮询
  start() {
    this.isRunning = true;
    this._poll();
  }

  // 停止长轮询
  stop() {
    this.isRunning = false;
  }

  // 核心轮询逻辑
  async _poll() {
    if (!this.isRunning) return;

    // 创建 AbortController 用于超时控制
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), this.timeout);

    try {
      const response = await fetch(this.url, {
        signal: controller.signal,
        headers: { 'Content-Type': 'application/json' }
      });

      clearTimeout(timeoutId);

      if (!response.ok) {
        throw new Error(`HTTP 错误: ${response.status}`);
      }

      const data = await response.json();

      // 收到正常数据，重置重试间隔
      this.currentRetryDelay = this.retryDelay;

      // 触发数据处理回调
      this.onMessage?.(data);
    } catch (error) {
      if (error.name === 'AbortError') {
        console.log('请求超时，将重新发起（这是正常的）');
      } else {
        console.error('长轮询出错:', error);
        // 出错时指数退避
        await this._backoff();
      }
    } finally {
      // 继续下一轮
      this._poll();
    }
  }

  // 指数退避策略
  async _backoff() {
    await delay(this.currentRetryDelay);
    // 每次重试间隔翻倍，但不超过最大值
    this.currentRetryDelay = Math.min(
      this.currentRetryDelay * 2,
      this.maxRetryDelay
    );
  }
}

// 使用示例
const client = new LongPollingClient('/api/notifications/poll', {
  timeout: 25000,
  retryDelay: 3000
});

client.onMessage = (data) => {
  console.log('收到通知:', data);
  renderNotification(data);
};

client.start();
```

##### 服务端实现（Node.js / Express）

```javascript
const express = require('express');
const app = express();

// 存储等待中的客户端请求（生产环境应使用 Redis 等）
const clients = new Map();

// 长轮询接口
app.get('/api/messages/poll', (req, res) => {
  const clientId = req.query.clientId || 'anonymous';

  // 将请求挂起，不立即响应
  clients.set(clientId, res);

  // 设置超时：30 秒后如果没有数据，返回空响应
  const timeoutId = setTimeout(() => {
    clients.delete(clientId);
    res.status(200).json({ messages: [], timeout: true });
  }, 30000);

  // 客户端断开连接时清理
  req.on('close', () => {
    clearTimeout(timeoutId);
    clients.delete(clientId);
  });
});

// 当有新消息时，推送给等待的客户端
app.post('/api/messages/send', express.json(), (req, res) => {
  const { clientId, message } = req.body;

  if (clients.has(clientId)) {
    const clientRes = clients.get(clientId);
    clients.delete(clientId);

    // 立即响应等待中的请求
    clientRes.status(200).json({ messages: [message] });
  }

  res.status(200).json({ success: true });
});

app.listen(3000);
```

#### 2. 进阶用法

##### 带"消息编号"的可靠长轮询

上面的基础实现有个问题：如果客户端错过了某些消息（比如断线期间），就再也收不到了。解决方案是**消息编号 + 断线续传**。

```javascript
/**
 * 带消息编号的长轮询客户端
 * 确保断线重连后能收到错过的消息
 */
class ReliableLongPollingClient {
  constructor(url) {
    this.url = url;
    this.lastMessageId = 0; // 记录已收到的最后一条消息编号
    this.isRunning = false;
  }

  async start() {
    this.isRunning = true;

    while (this.isRunning) {
      try {
        // 带上已收到的最后消息编号
        const response = await fetch(
          `${this.url}?lastId=${this.lastMessageId}`,
          { signal: AbortSignal.timeout(35000) }
        );

        if (!response.ok) throw new Error(`HTTP ${response.status}`);

        const { messages } = await response.json();

        for (const msg of messages) {
          // 只处理编号更大的消息（避免重复）
          if (msg.id > this.lastMessageId) {
            this.onMessage?.(msg);
            this.lastMessageId = msg.id;
          }
        }
      } catch (error) {
        console.error('轮询出错，稍后重试:', error);
        await delay(5000);
      }
    }
  }

  stop() {
    this.isRunning = false;
  }
}

// 使用
const client = new ReliableLongPollingClient('/api/messages/poll');
client.onMessage = (msg) => console.log('新消息:', msg);
client.start();
```

##### 服务端配合消息编号

```javascript
// 消息存储（生产环境用数据库）
const messageStore = [];
let nextId = 1;

// 长轮询接口：返回 lastId 之后的所有新消息
app.get('/api/messages/poll', (req, res) => {
  const lastId = parseInt(req.query.lastId) || 0;

  // 先检查是否有已存在的新消息（lastId 之后的）
  const newMessages = messageStore.filter(m => m.id > lastId);

  if (newMessages.length > 0) {
    // 有新消息，立即返回
    return res.json({ messages: newMessages });
  }

  // 没有新消息，挂起请求等待
  const pollHandler = {
    timeoutId: setTimeout(() => {
      cleanup();
      res.json({ messages: [] });
    }, 30000),
    res
  };

  // 存储等待中的请求
  pendingRequests.push(pollHandler);

  req.on('close', cleanup);

  function cleanup() {
    clearTimeout(pollHandler.timeoutId);
    const idx = pendingRequests.indexOf(pollHandler);
    if (idx !== -1) pendingRequests.splice(idx, 1);
  }
});

// 发送消息时，推送给所有等待中的客户端
app.post('/api/messages/send', express.json(), (req, res) => {
  const message = {
    id: nextId++,
    content: req.body.content,
    timestamp: Date.now()
  };

  messageStore.push(message);

  // 唤醒所有等待中的客户端
  pendingRequests.forEach(handler => {
    clearTimeout(handler.timeoutId);
    handler.res.json({ messages: [message] });
  });
  pendingRequests = [];

  res.json({ success: true, id: message.id });
});
```

### 四、实际应用场景

#### 1. 即时聊天室的离线消息推送

```javascript
// 聊天室长轮询：获取未读消息
class ChatLongPolling {
  constructor(roomId) {
    this.roomId = roomId;
    this.lastSeenId = localStorage.getItem(`chat:${roomId}:lastId`) || 0;
  }

  async poll() {
    const response = await fetch(
      `/api/rooms/${this.roomId}/messages?after=${this.lastSeenId}`
    );
    const { messages } = await response.json();

    for (const msg of messages) {
      appendMessageToUI(msg);
      this.lastSeenId = msg.id;
      // 持久化最后消息编号，刷新页面后能续传
      localStorage.setItem(`chat:${this.roomId}:lastId`, msg.id);
    }

    // 立即发起下一轮
    this.poll();
  }
}
```

#### 2. 订单状态实时更新

```javascript
// 电商场景：轮询订单状态变化
async function pollOrderStatus(orderId) {
  while (true) {
    const { status } = await fetch(`/api/orders/${orderId}/status`).then(r => r.json());

    updateOrderUI(status);

    // 订单已终态，停止轮询
    if (['completed', 'cancelled', 'refunded'].includes(status)) {
      break;
    }

    // 否则继续挂起等待下一次状态变化
    await longPollStatus(orderId);
  }
}
```

#### 3. 后台任务进度通知

```javascript
// 大文件导出场景：轮询任务进度
async function pollTaskProgress(taskId) {
  const response = await fetch(`/api/tasks/${taskId}/progress`);
  const { progress, status, downloadUrl } = await response.json();

  updateProgressBar(progress);

    // 任务未完成，继续轮询
```

### 五、常见问题与注意事项

#### 1. 连接管理问题

| 问题 | 说明 | 解决方案 |
| --- | --- | --- |
| **连接数耗尽** | 浏览器对同一域名有并发连接限制（通常 6 个） | 长轮询会长期占用一个连接，避免同时开多个长轮询 |
| **服务器资源占用** | 每个挂起的请求都占用一个线程/文件描述符 | 使用异步 I/O（Node.js、Nginx）或消息队列 |
| **断网重连风暴** | 网络恢复后大量客户端同时重连 | 加入随机抖动（jitter）分散重连时间 |

#### 2. 超时设置建议

```javascript
// ❌ 超时太短：频繁断开重连，变成"短轮询"
const badTimeout = 2000;

// ❌ 超时太长：连接出错时反应迟钝
const badTimeout2 = 300000;

// ✅ 推荐：25~35 秒（略小于服务器的 30 秒 keep-alive 超时）
const goodTimeout = 30000;
```

> 💡 **提示：** 超时时间要综合考虑：服务器 keep-alive 超时、代理服务器超时（Nginx 默认 60 秒）、用户体验。通常设置在 **25~35 秒** 之间。

#### 3. 避免重连风暴

```javascript
// ✅ 加入随机抖动，避免大量客户端同时重连
function getRetryDelay(baseDelay) {
  const jitter = Math.random() * 2000; // 0~2000ms 的随机值
  return baseDelay + jitter;
}
```

#### 4. 内存泄漏防范

```javascript
// ⚠️ 注意：页面卸载前要停止轮询，否则会导致内存泄漏
window.addEventListener('beforeunload', () => {
  pollingClient.stop();
});

// 使用 AbortController 可以取消进行中的请求
const controller = new AbortController();
fetch('/api/poll', { signal: controller.signal });
// 需要取消时：
controller.abort();
```

### 六、与相关概念对比

#### 1. 短轮询 vs 长轮询

| 对比维度 | 短轮询 | 长轮询 |
| --- | --- | --- |
| **请求频率** | 固定间隔（如每 2 秒一次） | 收到响应后立即发起下一个 |
| **实时性** | 平均延迟 = 间隔时间 / 2 | 接近实时（有数据立即响应） |
| **服务器压力** | 大量无意义的空请求 | 连接占用时间长，但请求次数少 |
| **实现复杂度** | 简单 | 需要处理连接挂起和超时 |

#### 2. 长轮询 vs SSE vs WebSocket

这是三种主流的"服务器推送"方案，选择取决于具体场景：

| 对比维度 | 长轮询 | SSE（EventSource） | WebSocket |
| --- | --- | --- | --- |
| **通信方向** | 客户端发起，服务器响应 | 单向：服务器→客户端 | **全双工**双向通信 |
| **协议** | HTTP/1.1 | HTTP/1.1 | 独立协议（基于 HTTP 握手） |
| **实时性** | 接近实时 | 接近实时 | 真正实时 |
| **连接复用** | 每次请求都要重新建连 | 保持一条长连接持续推送 | 一条连接双向复用 |
| **断线重连** | 需手动实现 | 浏览器自动重连 | 需手动实现 |
| **二进制数据** | 支持 | 仅文本 | 支持 |
| **浏览器兼容** | 所有浏览器 | 除 IE 外都支持 | 现代浏览器都支持 |
| **适用场景** | 简单通知、兼容老浏览器 | 新闻推送、股票行情、日志流 | 聊天室、在线游戏、协同编辑 |

#### 3. 如何选择

```
需要双向通信（如聊天）？
  └─ 是 → WebSocket
  └─ 否 → 只需要服务器推送？
            └─ 是 → 浏览器都支持，不需要兼容 IE？
            │         └─ 是 → SSE（更简单、自动重连）
            │         └─ 否 → 长轮询（兼容性最好）
            └─ 否 → 普通请求就够了
```

### 七、总结

**长轮询的核心要点回顾：**

- **本质**：用 HTTP 模拟服务器推送，通过"挂起连接"减少空请求
- **流程**：请求 → 服务器等待 → 有数据后响应 → 客户端立即重新发起请求
- **优点**：兼容性好、实现简单、能穿透大多数防火墙和代理
- **缺点**：不如 WebSocket 实时、服务器需维持大量挂起连接
- **最佳场景**：需要服务器推送但不需要双向通信，且对实时性要求不是毫秒级的场景

> 💡 **提示：** 在新项目中，如果不需要兼容 IE，**SSE 通常是比长轮询更好的选择**——它基于标准 HTTP、支持自动重连、API 更简洁。长轮询的价值主要体现在**兼容老系统**和**不支持 WebSocket/SSE 的特殊环境**中。

### 八、完整示例：Vue 3 中的长轮询

下面是一个在 Vue 3 组合式 API 中使用长轮询的完整示例：

```vue
<!-- NotificationCenter.vue -->
<template>
  <div class="notification-center">
    <h3>通知中心</h3>
    <ul>
      <li v-for="item in notifications" :key="item.id">
        {{ item.content }}
      </li>
    </ul>
    <p v-if="status === 'connecting'">连接中...</p>
    <p v-if="status === 'error'" class="error">连接异常，正在重试...</p>
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';

const notifications = ref([]);
const status = ref('connecting');
let abortController = null;

/**
 * 启动长轮询获取通知
 */
async function startPolling() {
  abortController = new AbortController();

  while (true) {
    try {
      status.value = 'connecting';

      const response = await fetch(
        '/api/notifications/poll',
        { signal: abortController.signal }
      );

      if (!response.ok) throw new Error(`HTTP ${response.status}`);

      const data = await response.json();
      notifications.value.push(...data.notifications);
      status.value = 'connected';
    } catch (error) {
      if (error.name === 'AbortError') {
        // 主动停止，退出循环
        return;
      }
      // 出错时等 5 秒再重试
      status.value = 'error';
      await new Promise(r => setTimeout(r, 5000));
    }
  }
}

onMounted(() => {
  startPolling();
});

onUnmounted(() => {
  // 组件卸载时停止轮询
  abortController?.abort();
});
</script>
```
